    Sub LoadNextPage()
        If Left(mstrNodeKey,3) = "SYS" Then
            Call ExecuteAction(mstrAction, mstrNodeKey, 0, "")
        Else
            Select Case Left(mstrNodeKey,7)
                Case "POL|POL"
                    Call ExecuteAction("TREE", "POL|POL|" & mstrExpiredNumber, -1, "")
                Case Else
                    Call ExecuteAction(mstrAction, "POL|POL|" & mstrExpiredNumber, 0, "")
            End Select
        End If
    End Sub

STATUS: DONE (framework handles)
POC FILE: src/routes/routes.tsx + src/utils/data-strategy.ts
HOW: React Router runs parent→child loaders in order. The serial chain
     (menu.onload → LoadNextPage → tree.onload → LoadFirstPage) is
     IMPLICIT in the route tree. dataStrategy interleaves cycling.
     The SYS/POL|POL branch logic is in data-strategy.ts's
     shouldCallCycling + getActionMenuFrameName.
