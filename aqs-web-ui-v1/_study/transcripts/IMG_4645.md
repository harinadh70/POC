    Sub SetControlFocus(ByRef strMatchCode, ByRef objCallingWindow)
        '*****************************************************************************
        'PURPOSE:
        'This subroutine is used to set the focus to ANY control on ANY tab.  It is
        'advisable to use this anytime you are not sure the control that needs
        'focus is on the current tab. This routine will "look" for the control
        'on each tab, starting at the first tab.  When the control is found, its
        'parent tab will be "pressed" THEN focus will be set.
        '
        'NOTE: just send the MATCHCODE of the desired control to this function - NOT
        '           the actual control ID (dtaLPOLNUM) OR a reference to a control.
        '
        'PARAMETERS:
        ' strMatchCode: The matchcode of the control at which focus will be set.
        '
        '*****************************************************************************
        On Error Resume Next

        Dim objControl
                                        'A handle to the control at which we want focus.
        Dim colDIVs
                                        'Collection of all divTab tags (these are the tabs).
        Dim intTab
                                        'Tab counter.
        Dim intTabCount
                                        'Total number of tabs.


        With objCallingWindow
            '--These are the containers (tab pages) for all the editable controls--

            Set colDIVs = .document.getElementsByName("divTab")
            Call TrapBrowserError("EEBrowser|SetControlFocus|1")
            intTabCount = colDIVs.length - 1
