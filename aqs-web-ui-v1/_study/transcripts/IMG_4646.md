            Set colDIVs = .document.getElementsByName("divTab")
            Call TrapBrowserError("EEBrowser|SetControlFocus|1")
            intTabCount = colDIVs.length - 1

            If intTabCount < 0 Then
                intTabCount = 0
            Else
                intTabCount = colDIVs.length - 1
            End If

            For intTab = 0 To intTabCount
                '--Try to get a handle to the desired control through the tab <DIV>--
                If intTabCount = 0 Then
                    '--Since there is only one tab, we can get the control directly--
                    Set objControl = .document.all.item("dta" & strMatchCode)
                        Call TrapBrowserError("EEBrowser|SetControlFocus|2")
                Else
                    '--There are two or more tabs, so we need to look for the desired--
                    '--control THROUGH the tab object so we know what tab it is on.  --
                    'First, try to get a handle to the control the "normal" way.
                    Set objControl = colDIVs(intTab).all.item("dta" & strMatchCode)
                    If objControl Is Nothing Then
                        'If the "normal" way didn't work, try a different reference that works
                        'when the "divTab" DIV that is the desired tab is enclosed within another
                        'DIV tag that is used to position all the controls on a particular tab in
                        'unison.
                        If colDIVs(intTab).children.length > 0 Then
                            Set objControl = colDIVs(intTab).children.item.all("dta" & strMatchCode)
                        End If
                    End If

                    Call TrapBrowserError("EEBrowser|SetControlFocus|3")
                End If

                If Not objControl Is Nothing Then
                    'if the control with the matchcode is found...
                    If Not .mblnModal Then
                        '--Only do this if we are NOT in a modal screen--
                        If CInt(.tabstrip.SelectedTab) <> intTab Then
                            .tabstrip.SelectedTab = intTab
                                'select the tab containing the control
                        End If
