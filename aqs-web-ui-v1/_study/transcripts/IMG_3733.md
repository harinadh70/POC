export function buildModalUrl(
    url: string,
    width: string = '600',
    height: string = '400',
    queryString?: string,
): string {
    const params = new URLSearchParams();
    params.set('url', url);
    params.set('width', width);
    params.set('height', height);

    if (queryString) {
        params.set('query', queryString);
    }

    return `/modal-renderer?${params.toString()}`;
}
