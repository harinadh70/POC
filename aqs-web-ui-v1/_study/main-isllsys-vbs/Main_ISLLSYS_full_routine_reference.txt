Purpose: Boot routine for the frameset shell page. Sets up global references, decides whether this
is a policy / search / utility shell, sizes the frame rows/cols, loads the security data island,

      On Error Resume Next
      AddWindowToCollection window
      Set GlobalVars = New globalVarsClass
      Set mobjAQSMain = window
      Set mobjParentWindow = window.opener
      If mblnReplacementHomePage Then ... rebuild child-window refs ...
      document.body.setAttribute "policyid", mstrPolicyID
      If CLng(mstrPolicyID) > 0 Then fraHorizontal.rows="46,*": fraTreeColumn.rows="90,*"
      ElseIf CLng(mstrPolicyID)=0 Then mblnSearchPage=True ...
      Set mxmlSecurity = CreateObject("MSXML2.DOMDocument.6.0")
      mxmlSecurity.loadXML(document.all("xdiSecurity").XMLDocument.xml)
      Call ExecuteAction("menu", "POL|POL|" & mstrExpiredNumber & "|", -1, "")
    End Sub

--- React POC  (engine/load/shell-load.ts) -----------------------------------------------------
    // engine/load/shell-load.ts (SHIPPED)
    export async function shellLoad(session) {
      const mode = resolveShellMode(session.policyId); // policy|search|utility
      appStore.setGlobalVars(session);          // GlobalVars + marrSessionInformation
      authStore.setSecurity(session.security);  // was the xdiSecurity island
      const action = resolveInitialAction(session); // POL|POL default, LIB for -3
      await executeAction('menu', `POL|POL|${session.expiredNumber}|`, -1);

    export function useShellLoad() { useEffect(() => { shellLoad(...); }, []); }

How it maps: This is the one routine in the section with real, working React code. shellLoad +

that chose policy vs. search vs. utility layout, resolveInitialAction replaces the mstrAction/-3
logic, and the security XML island is parsed once into auth-store instead of an MSXML DOMDocument.
The frame-sizing lines (fraHorizontal.rows, fraVertical.cols) are gone because the SPA uses CSS
layout, not a frameset. The kickoff call to ExecuteAction('menu', ...) is preserved as
executeAction, which begins the serial page load.

--- Legacy VBScript ------------------------------------------------------------------------------

--- React POC  (would live in: engine/load/shell-load.ts (+ config/routes.ts route loaders)) ---

    // The staggering hack disappears; ordering becomes explicit awaits / nested route loaders.
    if (nodeKey.startsWith('SYS')) {

6. AddWindowToCollection   [SCAFFOLDED]
   Sub · lines 516-528 · Windows
==================================================================================================
Purpose: Registers a window in a global Scripting.Dictionary shared across all app windows. It
walks up through openers to the top-most window that owns the collection, so every window can
later be found and closed (e.g. on logout) even if the home window is gone.

--- Legacy VBScript ------------------------------------------------------------------------------
    Sub AddWindowToCollection(myWindow)
      If Not IsEmpty(myWindow) Then
        If IsObject(window.opener) And Not IsEmptyEval("window.opener.AddWindowToCollection") Then
          window.opener.AddWindowToCollection myWindow      ' bubble up to top owner
        Else
          windowsCollection.Add CStr(windowsCollection.Count), myWindow
          Set myWindow.windowsCollection = windowsCollection ' share ref down to all windows
        End If
      End If
    End Sub

--- React POC  (engine/window-manager.ts) ------------------------------------------------------
    // engine/window-manager.ts (SCAFFOLDED registry)
    export function addWindow(win: AppWindow) {
      windowRegistry.set(win.key, win);   // one shared module-level registry
    }
    // No opener-walking: in an SPA there is a single JS context, so every
    // 'window' (modal / tab / panel) registers into the same in-memory map.

How it maps: The whole reason this recursively bubbles up through window.opener is that each
browser window in the old app has its own script context, so they must nominate one top window to
hold the master list and then share the Dictionary reference back down. A single-page React app
has one JS context, so the window-manager module already IS the single shared registry — the
opener-walking and reference-propagation collapse to a plain map insert. window-manager.ts exists
as a scaffolded registry keyed one-per-window, which is the right home; this specific add-and-
share behaviour maps onto it directly but the logout-time enumeration it feeds is not fully built.
